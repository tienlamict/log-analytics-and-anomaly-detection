   - Define helper `parseIntParam(r *http.Request, name string, def, min, max int) (int, error)`:
     - Get `r.URL.Query().Get(name)`; if empty return `def, nil`
     - Parse with `strconv.Atoi`; if err return `0, fmt.Errorf("%s: not an integer", name)`
     - Clamp: if v < min set v = min; if v > max set v = max
     - Return v, nil

   - Define helper `parseTimeParam(r *http.Request, name string) (time.Time, error)`:
     - Get `r.URL.Query().Get(name)`; if empty return `time.Time{}, nil`
     - Parse with `time.Parse(time.RFC3339, raw)`; if err return `time.Time{}, fmt.Errorf("%s: must be RFC 3339 format", name)`
     - Return t, nil

   - Define `func (s *Server) handleListLogs(w http.ResponseWriter, r *http.Request)`:
     - Parse query params:
       - `service := r.URL.Query().Get("service")`
       - `level := r.URL.Query().Get("level")`
       - `page, err := parseIntParam(r, "page", 1, 1, 10000)` — on error: `writeError(w, 400, err.Error()); return`
       - `size, err := parseIntParam(r, "size", 20, 1, 1000)` — on error: `writeError(w, 400, err.Error()); return`
       - `from, err := parseTimeParam(r, "from")` — on error: `writeError(w, 400, err.Error()); return`
       - `to, err := parseTimeParam(r, "to")` — on error: `writeError(w, 400, err.Error()); return`
     - Build `domain.LogQuery{Service: service, Level: level, From: from, To: to, Page: page, Size: size}`
     - Call `entries, total, err := s.logStore.SearchLogs(r.Context(), q)`
     - If err != nil: `s.logger.Error("search logs failed", zap.Error(err)); writeError(w, 500, "internal error"); return`
     - `writeJSON(w, 200, PagedResponse{Data: entries, Total: total, Page: page, Size: size})`

   - Define `func (s *Server) handleGetLog(w http.ResponseWriter, r *http.Request)`:
     - `id := chi.URLParam(r, "id")`
     - If `id == ""`: `writeError(w, 400, "missing id"); return`
     - `entry, err := s.logStore.GetLog(r.Context(), id)`
     - If err != nil and `errors.Is(err, domain.ErrNotFound)`: `writeError(w, 404, "log entry not found"); return`
     - If err != nil (other): `s.logger.Error("get log failed", zap.Error(err), zap.String("id", id)); writeError(w, 500, "internal error"); return`
     - `writeJSON(w, 200, entry)`
